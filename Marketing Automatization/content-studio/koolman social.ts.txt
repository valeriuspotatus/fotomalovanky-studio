export function getSelectedProducts(productIds: string[]): Product[] {
  return allProductItems.filter((p) => productIds.includes(p.id));
}

export function getSelectedConcerns(concernIds: string[]): Category[] {
  return categories.filter((c) => concernIds.includes(c.id));
}

export function getSelectedCategories(categoryIds: string[]): Category[] {
  return categories.filter((c) => categoryIds.includes(c.id));
}

export function getSelectedIngredients(ingredientIds: string[]): Ingredient[] {
  return Object.values(ingredients).filter((i) => ingredientIds.includes(i.id));
}

export function getSelectedEvents(eventIds: string[]): CalendarEvent[] {
  return eventIds.map((id) => getEventById(id)).filter((e): e is CalendarEvent => e !== undefined);
}

// Format context for prompts - DETAILED version with strict labeling
function formatProductContext(product: Product): string {
  if (product.type === 'set' || product.type === 'giftbox') {
    const contentProducts =
      product.contents?.map((pid) => products.find((p) => p.id === pid)).filter((p): p is Product => p !== undefined) ||
      [];

    const contentsList = contentProducts.map((p) => p.title.cs).join(', ');
    const allIngredients = new Set<string>();
    const allBenefits = new Set<string>();
    const allConcerns = new Set<string>();

    contentProducts.forEach((p) => {
      p.ingredients.forEach((i) => allIngredients.add(i.name.cs));
      p.benefits.cs.forEach((b) => allBenefits.add(b));
      allConcerns.add(p.category);
    });

    return `
### ${product.title.cs} (${product.type === 'set' ? 'Výhodný set' : 'Dárkový box'})
**OBSAHUJE PRODUKTY:** ${contentsList}
**VŠECHNY INGREDIENCE V SETU:** ${Array.from(allIngredients).join(', ') || 'Neuvedeno'}
**VŠECHNY BENEFITY:** ${Array.from(allBenefits).join('; ')}
**POPIS SETU:** ${product.description.cs}
**PRO KOHO:** ${product.forWhom?.cs?.join(', ') || 'Pro každého muže'}
**CONCERNS (typy problémů):** ${Array.from(allConcerns).join(', ') || 'Obecné'}
**TAGLINE:** "${product.tagline?.cs || ''}"
`;
  }

  // Regular individual product - with strict labeling
  const ingredientsList = product.ingredients.map((i) => i.name.cs).join(', ') || 'Neuvedeno';
  const benefitsList = product.benefits.cs.join('; ');
  const whenToUseList = product.whenToUse?.cs?.join(', ') || 'Denně';
  const concernsList = product.category || 'Obecné';

  return `
### ${product.title.cs}
**⚠️ POUZE TYTO INGREDIENCE (jiné NEZMIŇUJ):** ${ingredientsList}
**⚠️ POUZE TYTO BENEFITY:** ${benefitsList}
**PROBLÉM, KTERÝ ŘEŠÍ:** ${product.problemStatement?.cs || product.description.cs}
**ŘEŠENÍ:** ${product.solutionStatement?.cs || product.tagline?.cs || ''}
**KDY POUŽÍT:** ${whenToUseList}
**CONCERNS (typy problémů):** ${concernsList}
**TAGLINE:** "${product.tagline?.cs || ''}"
`;
}

function formatConcernContext(concern: Category): string {
  return `
### ${concern.title.cs}
- **Problém:** ${concern.problem.cs}
- **Řešení:** ${concern.solution.cs}
- **Doporučené produkty:** ${concern.recommendedProducts.join(', ')}
`;
}

function formatIngredientContext(ingredient: Ingredient): string {
  return `**${ingredient.name.cs}**
Benefit: ${ingredient.benefit.cs}`;
}

export function buildBrainstormSocialPrompt(input: SocialGeneratorInput): string {
  const selectedProducts = getSelectedProducts(input.productIds);
  const selectedConcerns = getSelectedConcerns(input.concernIds);
  const selectedCategories = getSelectedCategories(input.categoryIds);
  const selectedIngredients = getSelectedIngredients(input.ingredientIds);
  const selectedEvents = getSelectedEvents(input.eventIds);

  const platformInfo = PLATFORMS[input.platform];
  const formatInfo = OUTPUT_FORMATS[input.outputFormat];
  const journeyInfo = JOURNEY_STAGES[input.journeyStage];
  const styleInfo = CONTENT_STYLES[input.contentStyle];
  const seasonInfo = input.season ? SEASONS[input.season] : null;
  const contextInfo = input.context ? CONTEXTS[input.context] : null;

  // Build context sections based on what's selected
  const contextSections: string[] = [];

  if (selectedProducts.length > 0) {
    contextSections.push(`### Vybrané produkty:
${selectedProducts.map(formatProductContext).join('\n\n')}`);
  }

  if (selectedConcerns.length > 0) {
    contextSections.push(`### Vybrané problémy:
${selectedConcerns.map(formatConcernContext).join('\n\n')}`);
  }

  if (selectedCategories.length > 0) {
    contextSections.push(`### Vybrané kategorie:
${selectedCategories.map((c) => `**${c.title.cs}**: ${c.description.cs}`).join('\n')}`);
  }

  if (selectedIngredients.length > 0) {
    contextSections.push(`### Vybrané ingredience:
${selectedIngredients.map(formatIngredientContext).join('\n')}`);
  }

  if (selectedEvents.length > 0) {
    contextSections.push(`### Vybrané události/příležitosti:
${selectedEvents.map(formatEventContext).join('\n\n')}`);
  }

  // Season context
  if (seasonInfo) {
    contextSections.push(`### Sezónní kontext: ${seasonInfo.cs}
${seasonInfo.context}`);
  }

  // Situational context
  if (contextInfo) {
    contextSections.push(`### Situační kontext: ${contextInfo.cs}
${contextInfo.description}`);
  }

  // Content sources section with actual content
  const sourcesSection =
    input.contentSources.length > 0
      ? `### Content sources k využití:
${input.contentSources
  .map((sourceId) => {
    const sourceInfo = CONTENT_SOURCES[sourceId];
    const userContent = input.sourceContent?.[sourceId];
    if (userContent) {
      return `**${sourceInfo.cs}:**
${userContent}`;
    }
    return `- **${sourceInfo.cs}**: ${sourceInfo.description}`;
  })
  .join('\n\n')}`
      : '';

  return `Jsi social media expert pro českou pánskou kosmetiku KOOLMAN. Tvým úkolem je vymyslet nápady na obsah.

${selectedProducts.length > 0 ? STRICT_CONSTRAINTS : ''}

${TONE_OF_VOICE}

## PLATFORMA: ${platformInfo.cs}
${platformInfo.specs}
${platformInfo.bestPractices}

## FORMÁT: ${formatInfo.cs}
${formatInfo.structure}
${formatInfo.tips}

## CUSTOMER JOURNEY: ${journeyInfo.cs}
**Cíl:** ${journeyInfo.goal}
**Tone:** ${journeyInfo.tone}
**Vhodné typy obsahu:** ${journeyInfo.contentTypes.join(', ')}
**CTA:** ${journeyInfo.cta.join(', ')}

## STYL OBSAHU: ${styleInfo.cs}
${styleInfo.description}
${styleInfo.guidelines}
**Příklady:** ${styleInfo.examples.join(' | ')}

## KONTEXT PRO BRAINSTORMING

${contextSections.join('\n\n')}

${sourcesSection}

${input.customPrompt ? `### Dodatečné zadání:\n${input.customPrompt}` : ''}

## TVŮJ ÚKOL

Vymysli 10 nápadů na ${formatInfo.cs.toLowerCase()} pro ${platformInfo.cs}, které:
1. Odpovídají fázi "${journeyInfo.cs}" customer journey
2. Jsou ve stylu "${styleInfo.cs}" - dodržuj guidelines pro tento styl
3. Využívají vybraný kontext (produkty/problémy/události)
4. Jsou v duchu Koolman tone of voice
5. Mají jasný hook, který zaujme v prvních vteřinách

Pro každý nápad uveď:
- **hook**: První věta/fráze která zaujme (max 15 slov)
- **concept**: O čem to je, jaký je hlavní message (2-3 věty)
- **visualIdea**: Jak by měl vypadat vizuál nebo video (1-2 věty)
- **journeyFit**: Proč to pasuje do ${journeyInfo.cs} fáze (1 věta)

Odpověz ve formátu JSON:
{
  "ideas": [
    {
      "id": "1",
      "hook": "...",
      "concept": "...",
      "visualIdea": "...",
      "journeyFit": "..."
    }
  ]
}`;
}

export function buildGenerateSocialPrompt(input: GenerateSocialInput): string {
  const selectedProducts = getSelectedProducts(input.productIds);
  const selectedConcerns = getSelectedConcerns(input.concernIds);
  // Categories and ingredients are collected for potential future use but not currently in prompt
  getSelectedCategories(input.categoryIds);
  getSelectedIngredients(input.ingredientIds);
  const selectedEvents = getSelectedEvents(input.eventIds);

  const platformInfo = PLATFORMS[input.platform];
  const formatInfo = OUTPUT_FORMATS[input.outputFormat];
  const journeyInfo = JOURNEY_STAGES[input.journeyStage];
  const styleInfo = CONTENT_STYLES[input.contentStyle];
  const seasonInfo = input.season ? SEASONS[input.season] : null;
  const contextInfo = input.context ? CONTEXTS[input.context] : null;

  // Build context
  const contextSections: string[] = [];

  if (selectedProducts.length > 0) {
    contextSections.push(`### Produkty:
${selectedProducts.map(formatProductContext).join('\n')}`);
  }

  if (selectedConcerns.length > 0) {
    contextSections.push(`### Problémy:
${selectedConcerns.map(formatConcernContext).join('\n')}`);
  }

  if (selectedEvents.length > 0) {
    contextSections.push(`### Události:
${selectedEvents.map(formatEventContext).join('\n\n')}`);
  }

  // Season context
  if (seasonInfo) {
    contextSections.push(`### Sezónní kontext: ${seasonInfo.cs}
${seasonInfo.context}`);
  }

  // Situational context
  if (contextInfo) {
    contextSections.push(`### Situační kontext: ${contextInfo.cs}
${contextInfo.description}`);
  }

  // Content sources with actual content
  if (input.contentSources.length > 0) {
    const sourcesContent = input.contentSources
      .map((sourceId) => {
        const sourceInfo = CONTENT_SOURCES[sourceId];
        const userContent = input.sourceContent?.[sourceId];
        if (userContent) {
          return `**${sourceInfo.cs}:**
${userContent}`;
        }
        return `- **${sourceInfo.cs}**: ${sourceInfo.description}`;
      })
      .join('\n\n');
    contextSections.push(`### Content sources:\n${sourcesContent}`);
  }

  // Format-specific output instructions
  let outputInstructions = '';

  if (input.outputFormat === 'social-post') {
    outputInstructions = `
Vytvoř kompletní social post. Odpověz ve formátu JSON:
{
  "output": {
    "caption": "Kompletní caption včetně emoji a formátování",
    "hashtags": ["hashtag1", "hashtag2", "..."],
    "visualConcept": "Detailní popis vizuálu/fotky/grafiky",
    "postingTip": "Tip na nejlepší čas/den pro posting"
  }
}`;
  } else if (input.outputFormat === 'ad') {
    outputInstructions = `
Vytvoř kompletní reklamní copy. Odpověz ve formátu JSON:
{
  "output": {
    "headlines": ["Headline 1", "Headline 2", "Headline 3"],
    "primaryText": "Hlavní text reklamy (max 125 znaků pro FB)",
    "description": "Doplňkový text pod hlavní text",
    "visualConcept": "Detailní popis vizuálu/video konceptu",
    "targetAudience": "Popis cílové skupiny pro targeting",
    "ctaButton": "Doporučené CTA tlačítko (Shop Now, Learn More, etc.)"
  }
}`;
  } else if (input.outputFormat === 'video') {
    outputInstructions = `
Vytvoř kompletní video script. Odpověz ve formátu JSON:
{
  "output": {
    "hookScript": "Přesný script pro prvních 3 vteřin - slovo od slova",
    "fullScript": "Kompletní script s časovými značkami [0:00-0:03] Hook\\n[0:03-0:10] Problem...",
    "visualDirections": ["Záběr 1: ...", "Záběr 2: ...", "..."],
    "musicSuggestion": "Doporučený typ hudby/zvuku",
    "onScreenText": ["Text 1 (0:00-0:05)", "Text 2 (0:05-0:15)", "..."],
    "duration": "Doporučená délka videa"
  }
}`;
  }

  return `Jsi social media expert pro českou pánskou kosmetiku KOOLMAN. Tvým úkolem je vytvořit finální obsah.

${selectedProducts.length > 0 ? STRICT_CONSTRAINTS : ''}

${TONE_OF_VOICE}

## PLATFORMA: ${platformInfo.cs}
${platformInfo.specs}

## FORMÁT: ${formatInfo.cs}
${formatInfo.structure}

## CUSTOMER JOURNEY: ${journeyInfo.cs}
Cíl: ${journeyInfo.goal}
Doporučené CTA: ${journeyInfo.cta.join(', ')}

## STYL OBSAHU: ${styleInfo.cs}
${styleInfo.guidelines}

## VYBRANÝ NÁPAD

**Hook:** ${input.selectedIdea.hook}
**Koncept:** ${input.selectedIdea.concept}
**Vizuální nápad:** ${input.selectedIdea.visualIdea}

## KONTEXT

${contextSections.join('\n\n')}

${input.customPrompt ? `### Dodatečné zadání:\n${input.customPrompt}` : ''}

## TVŮJ ÚKOL

${outputInstructions}`;
}
